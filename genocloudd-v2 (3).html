ai_interpreter.py🤖 Claude API Integration
# AI-powered variant interpretation using Claude API
import anthropic
from dataclasses import dataclass

client = anthropic.Anthropic()  # uses ANTHROPIC_API_KEY env var

@dataclass
class Variant:
    gene: str; chrom: str; pos: int
    ref: str; alt: str; impact: str
    af: float; clinvar: str

class AIVariantInterpreter:
    SYSTEM = """You are a clinical genomics AI assistant.
    Given a genetic variant, provide:
    1. Clinical significance summary
    2. Associated diseases/conditions
    3. Actionability (is there a treatment?)
    4. Confidence level
    Keep responses concise and evidence-based."""

    def interpret(self, variant: Variant) -> str:
        prompt = f"""Interpret this variant:
        Gene: {variant.gene}
        Change: {variant.ref} → {variant.alt}
        Impact: {variant.impact}
        Allele Freq: {variant.af:.1%}
        ClinVar: {variant.clinvar}
        Position: {variant.chrom}:{variant.pos}"""

        msg = client.messages.create(
            model="claude-sonnet-4-20250514",
            max_tokens=500,
            system=self.SYSTEM,
            messages=[{"role": "user", "content": prompt}]
        )
        return msg.content[0].text

    def batch_interpret(self, variants: list[Variant]) -> dict:
        """Interpret all HIGH-impact variants"""
        results = {}
        for v in variants:
            if v.impact == "HIGH":
                results[v.gene] = self.interpret(v)
        return results

# Kubernetes Job — GATK Variant Calling
apiVersion: batch/v1
kind: Job
metadata:
  name: gatk-variant-caller
  namespace: bioinformatics
  labels:
    app: genocloudd
    component: variant-calling
spec:
  parallelism: 4        # Run 4 chromosomes in parallel
  completions: 24       # One per chromosome
  template:
    spec:
      containers:
      - name: gatk
        image: broadinstitute/gatk:4.4.0.0
        resources:
          requests:
            memory: "8Gi"
            cpu: "4"
          limits:
            memory: "16Gi"
            cpu: "8"
        env:
        - name: SAMPLE_ID
          valueFrom:
            fieldRef: {fieldPath: metadata.name}
        command: ["python", "/app/run_gatk.py"]
        volumeMounts:
        - name: ref-genome
          mountPath: "/ref"
      restartPolicy: OnFailure
      volumes:
      - name: ref-genome
        persistentVolumeClaim:
          claimName: hg38-pvc
k8s/ai-deployment.yaml⎈ AI Service Deployment
# Deploy AI Interpreter as a K8s microservice
apiVersion: apps/v1
kind: Deployment
metadata:
  name: ai-interpreter
  namespace: bioinformatics
spec:
  replicas: 3             # 3 AI pods for high availability
  selector:
    matchLabels: {app: ai-interpreter}
  template:
    spec:
      containers:
      - name: ai-service
        image: helixai/ai-interpreter:1.0
        ports:
        - containerPort: 8080
        env:
        - name: ANTHROPIC_API_KEY
          valueFrom:
            secretKeyRef:
              name: api-secrets
              key: anthropic-key    # stored in K8s Secret
        livenessProbe:
          httpGet: {path: /health, port: 8080}
          initialDelaySeconds: 10
